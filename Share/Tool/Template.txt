using System;
using System.Collections.Generic;
using MongoDB.Bson.Serialization.Attributes;

namespace ET
{
    [Config]
    public partial class (ConfigName)Category 
        : Singleton<(ConfigName)Category>, ISingletonAwake
    {
        [BsonElement("list")]
        private List<(ConfigName)> list = new();

        [BsonIgnore]
        private Dictionary<int, (ConfigName)> dict = new();

        public void Awake()
        {
            this.dict.Clear();
            foreach ((ConfigName) item in this.list)
            {
                this.dict[item.Id] = item;
            }
        }

        public (ConfigName) Get(int id)
        {
            if (!this.dict.TryGetValue(id, out (ConfigName) item))
            {
                throw new Exception(
                    "配置不存在: (ConfigName) id=" + id);
            }
            return item;
        }

        public bool Contain(int id)
        {
            return this.dict.ContainsKey(id);
        }

        public IReadOnlyDictionary<int, (ConfigName)> GetAll()
        {
            return this.dict;
        }
    }

    public partial class (ConfigName)
    {
(Fields)
    }
}
